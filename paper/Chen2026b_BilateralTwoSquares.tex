\begin{abstract}
Let $N\equiv 2\pmod 4$ be sufficiently large. We show that the number of $n\le N$ such that $n$
and $N-n$ are both sums of two squares and $\Omega(n(N-n))\le 11$ is at least
$\frac32\,\Sg(N)N/(\log N)^2$, where $\Sg(N)$ is the binary Goldbach singular series. The level of distribution
of the sequence $n(N-n)$ is $1$, but sifting the primes $q\equiv 3\pmod 4$ up to $N^{1/2}$ with the linear
sieve falls exactly on the sieving limit $s=2$. We therefore sift these primes only up to $N^{1/2-\delta}$
and remove the survivors with two large prime factors $\equiv 3\pmod 4$ on the same side by a switching
argument at the Bombieri--Vinogradov level. The final numerical inequality is verified with interval
arithmetic. The same method gives $\max(\Omega(n),\Omega(N-n))\le 8$ for a positive proportion of the
expected number of $n$, and extends to the principal forms of all nine imaginary quadratic fields of class
number one, with the bound $11$ whenever $N$ is not divisible by a high power of the ramified prime.
\end{abstract}

\section{Introduction}

Every positive integer is a sum of four squares, so every $N$ is of the form $n_1+n_2$ with $n_1,n_2$
sums of two squares. We ask for such representations in which $n_1$ and $n_2$ have few prime factors.

The multiplicative analogue, pairs $n$, $n+1$ that are both sums of two squares, was settled up to
constants by Hooley \cite{Hooley} and Indlekofer \cite{Indlekofer}, who showed that there are
$\asymp x/\log x$ such $n\le x$; correlations $\sum r(n)r(n+h)$ of the representation function have since
been studied by spectral methods, with uniformity in the coefficients of the forms, e.g.\ by
Blomer \cite{Blomer}. On the side of almost prime variables, Br\"udern and Fouvry \cite{BF} proved
Lagrange's four squares theorem with almost prime variables, and Blomer, Grimmelt, Li and Rydin Myerson
\cite{BGLR} showed that every large $n\equiv1,3\pmod 6$ is a prime plus two squares of almost primes. In
these works the almost prime condition is on the variables $x_i$. Here it is on the values $x^2+y^2$
themselves. Finally, \cite{Chen1} treats the case where one summand is a prime and the other is an almost
prime sum of two squares; the present problem is its two-sided analogue.

\begin{theorem}\label{thm:main}
For every sufficiently large $N\equiv 2\pmod 4$,
\begin{multline*}
\#\{n\le N:\ n\equiv 1\ (\mathrm{mod}\ 4),\ n=x^2+y^2,\ N-n=z^2+w^2,\ \Omega(n(N-n))\le 11\}\\
\ge\ \frac32\,\Sg(N)\,\frac{N}{(\log N)^2},
\end{multline*}
where $\Sg(N)=2\prod_{\ell>2}\bigl(1-(\ell-1)^{-2}\bigr)\prod_{\ell\mid N,\,\ell>2}\frac{\ell-1}{\ell-2}$.
\end{theorem}

\begin{remark}[The prime $2$]
The restriction $N\equiv2\pmod4$ is natural. The odd part of a sum of two squares is $\equiv1\pmod 4$.
If $N\equiv0\pmod 4$, both summands must contain powers of $2$, and for $N=2^j$ the only representation is
$n_1=n_2=2^{j-1}$, so $\Omega(n_1n_2)$ is unbounded. For general $N$ one should count prime factors of the
odd parts only; we do not pursue this here.
\end{remark}

\begin{remark}
With the parameters of the second line of Table~\ref{tab:cert} the same method gives the weaker bound
$\Omega(n(N-n))\le 12$ with the larger constant $2.24$ in place of $\frac32$; both are certified.
\end{remark}

The weights of Section~\ref{sec:switch} also control each side separately.

\begin{corollary}\label{cor:side}
For every sufficiently large $N\equiv2\pmod4$, the number of $n\le N$, $n\equiv1\pmod 4$, such that $n$ and
$N-n$ are sums of two squares, $\max(\Omega(n),\Omega(N-n))\le 8$ and $\Omega(n(N-n))\le12$, is at least
$\frac35\,\Sg(N)N/(\log N)^2$.
\end{corollary}

The method is not specific to $x^2+y^2$. Let $\Delta<0$ be a fundamental discriminant with class number
$h(\Delta)=1$, that is $\Delta\in\{-3,-4,-7,-8,-11,-19,-43,-67,-163\}$, let $Q_\Delta$ be the principal
binary quadratic form of discriminant $\Delta$, and let $r$ be the prime ramified in $\mathbb Q(\sqrt\Delta)$.

\begin{theorem}\label{thm:general}
Let $N$ be a sufficiently large even integer. Then
\[
\#\{n\le N:\ n=Q_\Delta(x,y),\ N-n=Q_\Delta(z,w),\ \Omega(n(N-n))\le k_\Delta(N)\}
\ \gg_\Delta\ \Sg(N)\,\frac{N}{(\log N)^2},
\]
where $k_\Delta(N)=11$ in each of the following cases:
$\Delta=-4$ and $N\equiv2\ (\mathrm{mod}\ 4)$;
$\Delta=-8$ and $v_2(N)\le2$;
$\Delta=-3$ and $N\equiv2\ (\mathrm{mod}\ 3)$;
$\Delta=-r$ with $r\ge7$ and $r\nmid N$.
Moreover $k_{-3}(N)=12$ if $N\equiv1\pmod 3$, and $k_\Delta(N)=13$ in the cases
$\Delta=-4$, $N\equiv4\ (\mathrm{mod}\ 8)$;
$\Delta=-8$, $v_2(N)=3$;
$\Delta=-3$, $N\equiv6\ (\mathrm{mod}\ 9)$;
$\Delta=-r$ with $r\ge7$ and $r\,\|\,N$.
\end{theorem}

The extra prime factors in the second list are forced: they are powers of the ramified prime that every
representation must contain (Lemma~\ref{lem:admissible}). The proof is given in Section~\ref{sec:general}.

\subsection*{Method}
Restrict to $n\equiv1\pmod4$; then $N-n\equiv1\pmod4$. If neither $n$ nor $N-n$ has a prime factor
$q\equiv3\pmod 4$, both are sums of two squares. The sequence $\{n(N-n)\}$ has level of distribution
$N^{1-\varepsilon}$ (Lemma~\ref{lem:level}), and the primes $q\equiv3\pmod4$ have two roots of
$n(N-n)\equiv0\pmod q$, so they form a sifting set of dimension $1$. Sifting them up to $N^{1/2}$ at level
$N$ gives $s=2$, where the linear lower bound function vanishes. We sift them up to $y=N^{1/2-\delta}$ instead.
Since $n\equiv N-n\equiv1\pmod 4$, each side has an even number of such prime factors, so a survivor that is
not a pair of sums of two squares has two large prime factors $\equiv3\pmod4$ on the \emph{same} side; these
are counted from above by switching (Section~\ref{sec:switch}). Prime factors are controlled by Richert's
weights applied to $n(N-n)$ jointly.

\begin{remark}
To the best of the author's knowledge, Theorems~\ref{thm:main} and~\ref{thm:general} have not appeared in
the literature.
\end{remark}

\section{The sequence $n(N-n)$}\label{sec:level}

Let $N\equiv 2\pmod 4$ be large, $\cP_1=\{q\equiv1\pmod4\}$, $\cP_3=\{q\equiv3\pmod4\}$, and
\[
\cA=\{\,a_n=n(N-n):\ 1\le n\le N,\ n\equiv1\ (\mathrm{mod}\ 4)\,\},\qquad X=N/4.
\]
For odd squarefree $d$ let $\rho(d)$ be the number of solutions of $n(N-n)\equiv0\pmod d$; it is
multiplicative with $\rho(q)=2$ for $q\nmid N$ and $\rho(q)=1$ for $q\mid N$. Put $g(d)=\rho(d)/d$, so that
\[
g(q)=\frac2q\quad(q\nmid N),\qquad g(q)=\frac1q\quad(q\mid N),
\]
and define $r_d$ by $\#\{n:\ a_n\in\cA,\ d\mid a_n\}=Xg(d)+r_d$.

\begin{lemma}[Level of distribution]\label{lem:level}
$|r_d|\le\rho(d)$, and for $D\le N$,
$\sum_{d\le D}\mu^2(d)\tau_3(d)|r_d|\ll D(\log D)^{C}$. In particular the level of distribution is
$N^{1-\varepsilon}$.
\end{lemma}

\begin{proof}
The condition $d\mid a_n$ together with $n\equiv1\pmod 4$ restricts $n$ to $\rho(d)$ classes modulo $4d$,
each containing $N/(4d)+O(1)$ integers $n\le N$. Hence $|r_d|\le\rho(d)\le 2^{\omega(d)}$, and the sum is
$\ll\sum_{d\le D}\tau_3(d)2^{\omega(d)}\ll D(\log D)^{C}$.
\end{proof}

\begin{lemma}[Local factors]\label{lem:local}
As $z\to\infty$,
\[
\prod_{2<q<z}(1-g(q))\sim\frac{\Sg(N)}{2}\Bigl(\frac{2e^{-\gamma}}{\log z}\Bigr)^2 .
\]
\end{lemma}

\begin{proof}
For $q\nmid N$, $(1-2/q)(1-1/q)^{-2}=1-(q-1)^{-2}$; for $q\mid N$, $(1-1/q)(1-1/q)^{-2}=q/(q-1)$. The product
of these factors over odd $q$ converges to $\Sg(N)/2$, and $\prod_{2<q<z}(1-1/q)\sim 2e^{-\gamma}/\log z$.
\end{proof}

We use the following sieve tools; see \cite{Chen1} for statements and references.
The linear sieve ($\kappa=1$, $\beta=2$) has functions $F_1$, $f_1$ with $F_1(s)=2e^\gamma/s$ for $0<s\le3$ and
$f_1(s)=2e^{\gamma}\log(s-1)/s$ for $2\le s\le 4$, $f_1(s)=0$ for $s\le2$
\cite{IwaniecRosser}; the semi-linear sieve ($\kappa=1/2$, $\beta=1$) has functions $F_{1/2}$,
$f_{1/2}$ \cite{IwaniecHalf}. The vector sieve inequality \cite{BF} is
$ab\ge a^-b^++a^+b^--a^+b^+$ for $a,b\in\{0,1\}$, $a^-\le a\le a^+$, $b^-\le b\le b^+$, $a^+,b^+\ge0$.
The Bombieri--Vinogradov theorem for convolutions is used as in \cite[Lemma~2.4]{Chen1}.

Figure~\ref{fig:linear} shows $F_1$ and $f_1$. Sifting the primes $q\equiv3\pmod4$ up to $N^{1/2}$ at
level $N$ would give $s=2$, where $f_1$ vanishes; with $y=N^{1/2-\delta}$ the working point moves to
$s_I=\theta_I/(1/2-\delta)$, which is $\approx3.62$ for the parameters of Theorem~\ref{thm:main}.

\section{The weighted sieve and switching}\label{sec:switch}

Fix $0<\alpha<\delta<1/4$, $0<\lambda$, $\alpha<\beta_1\le1/2$, and put
\[
y=N^{1/2-\delta},\quad z=N^{\alpha},\quad y_1=N^{\beta_1},\quad
P_I=\prod_{\substack{q\in\cP_3\\ q<y}}q,\quad P_S=\prod_{\substack{q\in\cP_1\\ q<z}}q,\quad P=P_IP_S .
\]
Every odd prime $q<z$ divides $P$. For $a_n\in\cA$ with $(a_n,P)=1$ define
\[
w_n=1-\lambda\sum_{\substack{q\mid a_n,\ q\in\cP_1\\ z\le q<y_1}}\Bigl(1-\frac{\log q}{\log y_1}\Bigr),
\qquad W=\sum_{(a_n,P)=1}w_n .
\]

\begin{lemma}[Structure]\label{lem:structure}
Let $(a_n,P)=1$. Then either
\begin{enumerate}
\item[(i)] $n$ and $N-n$ are both sums of two squares, all their prime factors are $\ge z$, and those
in $\cP_3$ occur only as squares $q^2$; or
\item[(ii)] for $m'=n$ or $m'=N-n$ we have $m'=q_1q_2m$ with $q_1<q_2$, $q_1,q_2\in\cP_3$, $q_1\ge y$, and
$m\le N^{2\delta}$ composed of primes of $\cP_1$ that are $\ge z$.
\end{enumerate}
\end{lemma}

\begin{proof}
Each of $n$, $N-n$ is odd and $\equiv1\pmod4$, so each has an even number of prime factors in $\cP_3$,
counted with multiplicity. They are all $\ge y$, and $y^4>N$, so each side has $0$ or $2$ of them. If on both
sides there are none, or they coincide, both sides are sums of two squares. Otherwise some side is as in
(ii). In particular, two exceptional prime factors never lie on different sides.
\end{proof}

\begin{lemma}[Prime factors]\label{lem:omega}
If $a_n$ is as in Lemma~\ref{lem:structure}(i), squarefree, and $w_n>0$, then
$\Omega(n(N-n))<1/\lambda+2/\beta_1$ and $\max(\Omega(n),\Omega(N-n))<1/\lambda+1/\beta_1$.
\end{lemma}

\begin{proof}
All prime factors of $a_n$ are in $\cP_1$ and $\ge z$, and since $a_n$ is squarefree each of them divides
exactly one of $n$, $N-n$. Write $w_n=1-\lambda(T_1+T_2)$, where $T_1$, $T_2$ are the sums over $q\mid n$ and
$q\mid N-n$. Each $T_i$ is a sum of non-negative terms, and as in \cite[Lemma~3.2]{Chen1},
$T_1\ge\Omega(n)-\log n/\log y_1\ge\Omega(n)-1/\beta_1$, and similarly for $T_2$. Now $w_n>0$ gives
$T_1+T_2<1/\lambda$, which implies both $\Omega(n(N-n))-2/\beta_1<1/\lambda$ and, since $T_2\ge0$,
$\Omega(n)-1/\beta_1<1/\lambda$; the same holds for $N-n$.
\end{proof}

Non-squarefree $a_n$ with a square factor $q^2$, $q\ge z$, number $\ll N^{1-\alpha}$ and are negligible.

\begin{proposition}[Main term]\label{prop:W}
Let $\theta_I\in(0,1)$ and, for $u\in[\alpha,\beta_1]$, $t(u)\in(0,1-u)$. Put $s_I=\theta_I/(1/2-\delta)$,
$s_S=(1-\theta_I)/\alpha$,
\[
L=f_1(s_I)F_1(s_S)+F_1(s_I)f_1(s_S)-F_1(s_I)F_1(s_S),
\]
\[
R=\int_\alpha^{\beta_1}\Bigl(1-\frac u{\beta_1}\Bigr)F_1\Bigl(\frac{t(u)}{1/2-\delta}\Bigr)
F_1\Bigl(\frac{1-u-t(u)}{\alpha}\Bigr)\frac{du}{u}.
\]
Then $W\ge(L-\lambda R+o(1))\,XV$, where $V=\prod_{q\mid P}(1-g(q))$ and
\[
XV\sim\frac{e^{-2\gamma}}{2\alpha(1/2-\delta)}\;\Sg(N)\frac{N}{(\log N)^2}.
\]
\end{proposition}

\begin{proof}
Apply the vector sieve with $a=1_{(a_n,P_I)=1}$ and $b=1_{(a_n,P_S)=1}$. Both are linear sieves:
$P_I$ and $P_S$ consist of primes from one half of the residue classes modulo $4$, each with
$\rho(q)=2$ (apart from the finitely many $q\mid N$ below a given bound, which do not affect the dimension).
Use levels $N^{\theta_I}$ and $N^{1-\theta_I-\varepsilon}$; the combined moduli are $\le N^{1-\varepsilon}$
and Lemma~\ref{lem:level} controls the remainders. For the weights, $q\mid a_n$ for $q\in\cP_1$, $q\nmid N$,
has density $g(q)=2/q$, and $\sum_{q\in\cP_1}2/q$ over $N^u\le q<N^{u+du}$ is $du/u$. The upper bound vector
sieve for $\cA_q$ at levels $N^{t(u)}$, $N^{1-u-t(u)-\varepsilon}$ gives $R$. Finally
$V=\prod_{2<q<z}(1-g(q))\prod_{q\in\cP_3,\,z\le q<y}(1-g(q))$, and Lemma~\ref{lem:local} together with
$\prod_{q\in\cP_3,\,z\le q<y}(1-2/q)\sim\log z/\log y$ gives the asymptotic for $XV$.
\end{proof}

\begin{proposition}[Switching]\label{prop:switch}
Let $\cE$ be the set of $n$ with $(a_n,P)=1$ in case (ii) of Lemma~\ref{lem:structure}. Then
\[
\#\cE\le\Bigl(4\,c(\delta,\alpha)\,e^{\gamma}\sqrt{\alpha(1/2-\delta)}\;U+o(1)\Bigr)\,XV,
\]
where $c(\delta,\alpha)$ is the switching density of \cite[Prop.~4.1]{Chen1} and
\[
U=\min_{0<t<1/2}F_{1/2}\Bigl(\frac{t}{1/2-\delta}\Bigr)F_{1/2}\Bigl(\frac{1/2-t}{\alpha}\Bigr).
\]
\end{proposition}

\begin{figure}[htbp]
\centering
\includegraphics[width=0.58\textwidth]{figure2_switching_domain.pdf}
\caption{The switching region for $\delta=0.245$, $\alpha=0.016$, in the coordinates $u_i=\log q_i/\log N$:
$\frac12-\delta\le u_1<u_2$ and $u_1+u_2\le1-\mu$, where $m=N^{\mu}$ with $\mu=0$ ($m=1$) or
$\mu\ge\alpha$. By Lemma~\ref{lem:structure} the triple $q_1q_2m$ lies entirely in $n$ (the set
$\cE_1$) or entirely in $N-n$ (the set $\cE_2$).}
\label{fig:switch}
\end{figure}

\begin{proof}
Write $\cE=\cE_1\cup\cE_2$ according as the exceptional pair lies in $n$ or in $N-n$; the two cases are
symmetric, which accounts for a factor $2$ (Figure~\ref{fig:switch}). Consider $\cE_1$. Let $\cT$ be the set of triples
$(q_1,q_2,m)$ as in Lemma~\ref{lem:structure}(ii) with $t=q_1q_2m\le N$, so that
$|\cT|=(c(\delta,\alpha)+o(1))N/\log N$ as in \cite{Chen1}; note $t\equiv1\pmod4$ automatically.
Every $n\in\cE_1$ equals such a $t$, and $N-t$ has no prime factor in $P$. Hence $\#\cE_1$ is at most the
number of elements of $\cB=\{N-t:\ t\in\cT\}$ coprime to $P$.

Let $\ell\mid P$. No prime factor of $t$ divides $P$ (the $q_i$ are $\ge y$ and the prime factors of $m$
are split and $\ge z$), so $t$ is equidistributed in the nonzero classes modulo $\ell$. Hence, for
$\ell\nmid N$, $\ell\mid N-t$ with density $g_\cB(\ell)=1/(\ell-1)$, while for $\ell\mid N$ we have
$\ell\nmid N-t$, so $g_\cB(\ell)=0$. The upper bound vector sieve for $\cB$ with the semi-linear components
$P_I$ and $P_S$ at total level $N^{1/2}(\log N)^{-B}$ (Bombieri--Vinogradov for the convolution
$q_1*(q_2m)$) gives
\[
\#\cE_1\le\bigl(U+o(1)\bigr)\,|\cB|\prod_{q\mid P}(1-g_\cB(q)).
\]
Now $(1-1/(\ell-1))(1-1/\ell)^{-1}=1-(\ell-1)^{-2}$ and $(1-0)(1-1/\ell)^{-1}=\ell/(\ell-1)$ for $\ell\mid N$:
these are the same local factors as in Lemma~\ref{lem:local}. Hence
\[
\prod_{q\mid P}(1-g_\cB(q))\sim\frac{\Sg(N)}{2}\cdot\frac{2e^{-\gamma}}{\log z}\Bigl(\frac{\log z}{\log y}\Bigr)^{1/2},
\]
and therefore
\[
\#\cE_1\le\bigl(U+o(1)\bigr)\,c\,\frac{N}{\log N}\cdot\frac{\Sg(N)e^{-\gamma}}{\alpha\log N}
\Bigl(\frac{\alpha}{1/2-\delta}\Bigr)^{1/2}
=\bigl(2c\,e^{\gamma}\sqrt{\alpha(1/2-\delta)}\,U+o(1)\bigr)XV,
\]
using Proposition~\ref{prop:W} for $XV$. The singular series cancels exactly. Adding $\cE_2$ gives the claim.
\end{proof}

\begin{lemma}\label{lem:numerics}
The bounds of Tables~\ref{tab:cert} and~\ref{tab:side} hold.
\end{lemma}

\begin{proof}[Method of verification]
The computation is carried out by the script \texttt{code/certify.py}.
\begin{enumerate}
\item The sieve functions $F_1,f_1$ ($\kappa=1$) and $F_{1/2},f_{1/2}$ ($\kappa=1/2$) are evaluated from their
closed forms on $(0,3]$, $[2,4]$ resp.\ $(0,2]$, $[1,3]$ in interval arithmetic (\texttt{mpmath.iv}, 80 bits).
Beyond these ranges the delay-differential equations $(s^\kappa F(s))'=\kappa s^{\kappa-1}f(s-1)$,
$(s^\kappa f(s))'=\kappa s^{\kappa-1}F(s-1)$ are integrated on the grid $10^{-3}\mathbb Z$ (up to $s=110$
resp.\ $60$): on each step the lagged function is enclosed between its values at the end points, using
that $F$ decreases and $f$ increases, and $\int\kappa s^{\kappa-1}ds$ is computed exactly. Each floating
point operation is followed by one outward rounding step. Off the grid, monotonicity gives one-sided bounds.
\item $L$ is bounded below via $L=f_1(s_I)F_1(s_S)+F_1(s_I)\bigl(f_1(s_S)-F_1(s_S)\bigr)$ and
$f_1(s_S)-F_1(s_S)\le0$.
\item $R$ is bounded above by a Riemann sum over $4000$ cells with $t(u)$ constant on each cell; on a cell
$[u_a,u_b]$ the factor $(1-u/\beta_1)/u$ is largest at $u_a$ and $F_1((1-u-t)/\alpha)$ is largest at $u_b$.
$U$ is bounded above by evaluating the product at one admissible $t$.
\item For $c$, the measure $\frac12\,1_{[\alpha,2\delta]}(\mu)\,d\mu/\mu$ is replaced by the masses
$\frac12\log\frac{k+1}{k}$ placed at the left end points $k/2000$. The integrand
$I(\mu)=\frac{1}{1-\mu}\log\frac{1/2+\delta-\mu}{1/2-\delta}$ is decreasing on $[0,2\delta]$: its derivative
has the sign of $\log\frac{a-\mu}{b}-\frac{1-\mu}{a-\mu}$ with $a=\frac12+\delta<1$, $b=\frac12-\delta$, whose
first term decreases and second increases, and which is negative at $\mu=0$ (checked in the script).
So moving mass to the left gives an upper bound. The floating point convolution is multiplied by $1+10^{-9}$.
\end{enumerate}
A finer independent computation of $c(0.22,0.01)$ (grid $1/20000$) gives $1.0913$, consistent with the
certified upper bound.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:main}]
Take the first line of Table~\ref{tab:cert}. By Lemma~\ref{lem:numerics} and Proposition~\ref{prop:W},
after removing the $\ll N^{1-\alpha}$ non-squarefree $a_n$, the number of counted $n$ is at least
\[
\bigl(0.038949+o(1)\bigr)\frac{e^{-2\gamma}}{2\cdot0.016\cdot0.255}\,\Sg(N)\frac{N}{(\log N)^2}
\ \ge\ \frac32\,\Sg(N)\frac{N}{(\log N)^2}
\]
for large $N$ (the numerical constant is $\ge1.5046$). Each of them has $n$ and $N-n$ sums of two squares
and $\Omega(n(N-n))\le 11$.
\end{proof}

\begin{proof}[Proof of Corollary~\ref{cor:side}]
The same argument with the parameters of Table~\ref{tab:side} gives at least
\[
\bigl(0.022031+o(1)\bigr)\frac{e^{-2\gamma}}{2\cdot0.022\cdot0.26}\,\Sg(N)\frac{N}{(\log N)^2}
\ \ge\ \frac35\,\Sg(N)\frac{N}{(\log N)^2}
\]
counted $n$ (the numerical constant is $\ge0.607$), and Lemma~\ref{lem:omega} gives $\max(\Omega(n),\Omega(N-n))\le8$ and $\Omega(n(N-n))\le12$.
\end{proof}

\section{Other class number one forms}\label{sec:general}

Let $\Delta$, $Q_\Delta$, $r$ be as in Theorem~\ref{thm:general} and $\chi=\chi_\Delta$ the Kronecker symbol.
An integer $m\ge1$ is represented by $Q_\Delta$ if and only if every prime $q$ with $\chi(q)=-1$ divides $m$
to an even power; the represented integers are the norms from $\mathcal O_{\mathbb Q(\sqrt\Delta)}$ and form a
multiplicative set containing $r$ \cite[Lemma~6.1]{Chen1}. For $u$ coprime to $\Delta$,
$\chi(u)=(-1)^{\#\{\text{inert prime factors of }u\}}$ (with multiplicity).

\begin{lemma}[Admissible classes]\label{lem:admissible}
Let $N$ be even and in one of the cases of Theorem~\ref{thm:general}, and let $\kappa_\Delta(N)=k_\Delta(N)-11$.
There are a modulus $M$, divisible by $2$ and by $r$, and a class $a\bmod M$ such that every $n\equiv a\pmod M$
satisfies $n=e\,u$, $N-n=e'u'$, where $e,e'$ are fixed powers of $r$ with $\Omega(ee')=\kappa_\Delta(N)$, the
integers $u,u'$ are coprime to $2r$, and $\chi(u)=\chi(u')=1$. Moreover, $r$ divides $M/e$ and $M/e'$.
\end{lemma}

\begin{proof}
Let $r$ be odd, so $\Delta=-r$ with $r\equiv3\pmod4$ and $\chi(u)=(u/r)$ for odd $u$. For $N'\not\equiv0\pmod r$,
\[
\#\{t\bmod r:\ (t/r)=((N'-t)/r)=1\}=\frac14\sum_{t\ne0,N'}\Bigl(1+\bigl(\tfrac tr\bigr)\Bigr)\Bigl(1+\bigl(\tfrac{N'-t}r\bigr)\Bigr)
=\frac{r-1-2(N'/r)}{4},
\]
since $\sum_t\bigl(\frac{t(N'-t)}r\bigr)=-\bigl(\frac{-1}r\bigr)=1$. This is $\ge1$ for $r\ge7$, and for $r=3$ it is
$1$ if $N'\equiv2$ and $0$ if $N'\equiv1\pmod 3$. If $r\nmid N$, take $N'=N$ and $e=e'=1$. If $r\,\|\,N$, then
$(n/r)=((N-n)/r)=1$ with $r\nmid n$ is impossible because $(-1/r)=-1$, so $r$ divides both sides; take $e=e'=r$
and $N'=N/r$. For $\Delta=-3$ and $N\equiv1\pmod3$ take $e=3$, $e'=1$, $n\equiv3\pmod9$; then
$u\equiv1$ and $N-n\equiv1\pmod 3$. For $\Delta=-3$, $N\equiv6\pmod9$ take $e=e'=3$ and $N'=N/3\equiv2$. In all
cases add the condition $n$ odd. For $\Delta=-4$: if $N\equiv2\pmod4$ take $n\equiv1\pmod4$; if $N\equiv4\pmod 8$
take $n\equiv2\pmod8$, so that $n/2\equiv N/2-n/2\equiv1\pmod4$. For $\Delta=-8$, where $\chi(u)=1$ iff
$u\equiv1,3\pmod 8$: if $v_2(N)\le2$ take $n\equiv1$ or $3\pmod 8$ with $N-n\equiv1$ or $3\pmod8$; if
$v_2(N)=3$ take $n\equiv2\pmod{16}$, so that $n/2\equiv1$ and $(N-n)/2\equiv3\pmod 8$. We have checked by
enumeration (script \texttt{code/local\_conditions.py}) that these are the minimal costs.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:general}]
Let $M,a,e,e'$ be as in Lemma~\ref{lem:admissible}, $\cA_\Delta=\{a_n=n(N-n):\ n\le N,\ n\equiv a\ (M)\}$ and
$X=N/M$. For $d$ coprime to $M$, $d\mid a_n$ restricts $n$ to $\rho(d)$ classes modulo $Md$, so
Lemma~\ref{lem:level} holds unchanged. We sift by the primes $q\nmid M$, with $\cP_3$, $\cP_1$ replaced by the
inert and split primes; both have density $1/2$ by Dirichlet's theorem modulo $|\Delta|$. Lemma~\ref{lem:structure}
holds for $u,u'$ (each has an even number of inert prime factors, all $\ge y$, and $y^4>N$), and
Lemma~\ref{lem:omega} applied to $uu'$ gives $\Omega(uu')\le11$, so $\Omega(n(N-n))\le 11+\kappa_\Delta(N)$.

It remains to check the normalisation. Put $\Sg_M(N)=\prod_{q\nmid M}(1-g(q))(1-1/q)^{-2}$. Since
$\prod_{q<z,\,q\nmid M}(1-1/q)\sim\frac{M}{\varphi(M)}\frac{e^{-\gamma}}{\log z}$,
\[
XV\sim\frac{M}{\varphi(M)^2}\,\Sg_M(N)\,\frac{e^{-2\gamma}}{\alpha(1/2-\delta)}\,\frac{N}{(\log N)^2}.
\]
In the switching step for the side $n=e\,t$, the triples $t=q_1q_2m$ satisfy $\chi(t)=1$ automatically and must
lie in one prescribed class modulo $M/e$ with $\chi=1$; by the prime number theorem in progressions they are
equidistributed among the $\varphi(M/e)/2$ such classes, and $t\le N/e$. Since $r\mid M/e$ we have
$e\,\varphi(M/e)=\varphi(M)$, so $|\cT|=(2/\varphi(M)+o(1))\,c(\delta,\alpha)N/\log N$. For $\ell\nmid M$, $\ell\mid N-et$
with density $1/(\ell-1)$ if $\ell\nmid N$ and $0$ if $\ell\mid N$; these are the local factors of $\Sg_M(N)$, and
\[
\prod_{q\mid P}(1-g_\cB(q))\sim\frac{M}{\varphi(M)}\,\Sg_M(N)\,\frac{e^{-\gamma}}{\log z}
\Bigl(\frac{\log z}{\log y}\Bigr)^{1/2}.
\]
Hence the contribution of this side is at most
$\bigl(2c\,e^{\gamma}\sqrt{\alpha(1/2-\delta)}\,U+o(1)\bigr)XV$, exactly as in Proposition~\ref{prop:switch},
and the same holds for the side $N-n=e't$. The Richert terms scale with $XV$ as before. Therefore the
normalised quantity is the same $\mathrm{net}$ as in Section~\ref{sec:numerics}, independently of $\Delta$ and
of the case, and Lemma~\ref{lem:numerics} gives a count $\ge(0.0389+o(1))XV\gg_\Delta\Sg(N)N/(\log N)^2$,
since $\Sg_M(N)$ and $\Sg(N)$ differ only in the local factors at the primes dividing $M$. For $\Delta=-4$,
$M=4$, one has $\frac{M}{\varphi(M)^2}\Sg_M(N)=\frac12\Sg(N)$, recovering Proposition~\ref{prop:W}.
\end{proof}
